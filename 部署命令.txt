# 部署视频分析API到Windows本地
# 复制以下命令到PowerShell执行

# 1. 创建目录
mkdir "C:\Users\Administrator\.openclaw\video-analysis\data"

# 2. 拉取Docker镜像
docker pull evil0ctal/douyin_tiktok_download_api

# 3. 启动容器
docker run -d --name douyin-api -p 18810:80 -v "C:\Users\Administrator\.openclaw\video-analysis\data:/app/data" --restart unless-stopped evil0ctal/douyin_tiktok_download_api

# 4. 验证容器运行
docker ps | findstr douyin-api

# 5. 测试API
# 打开浏览器访问: http://localhost:18810/docs

# 6. 配置Cookie
# 打开文件: C:\Users\Administrator\.openclaw\video-analysis\data\douyin_web\config.yaml
# 找到 Cookie: 这一行，替换为你的抖音cookie

# 7. 重启容器
docker restart douyin-api

# 8. 集成到OpenClaw Skill
# 修改: C:\Users\Administrator\.openclaw\skills\video-analysis\skill.ps1
# 将 API_URL 改为: http://localhost:18810
